def prim(graph, n):
    selected = [False] * n
    selected[0] = True

    edges = 0
    total_cost = 0

    while edges < n - 1:
        min_edge = 999
        x = y = 0

        for i in range(n):
            if selected[i]:
                for j in range(n):
                    if not selected[j] and graph[i][j]:
                        if graph[i][j] < min_edge:
                            min_edge = graph[i][j]
                            x, y = i, j

        print(x, "-", y, ":", graph[x][y])
        total_cost += graph[x][y]
        selected[y] = True
        edges += 1

    print("Total cost:", total_cost)


graph = [
    [0, 2, 0, 6, 0],
    [2, 0, 3, 8, 5],
    [0, 3, 0, 0, 7],
    [6, 8, 0, 0, 9],
    [0, 5, 7, 9, 0]
]

n = 5
prim(graph, n)
