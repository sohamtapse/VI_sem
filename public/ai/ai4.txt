def bound(board, row, col, n):
    # -------------------------------
    # BOUND FUNCTION (Branch & Bound)
    # -------------------------------

    # check column
    for i in range(row):
        if board[i][col] == 'Q':
            return False

    # check left diagonal
    i, j = row, col
    while i >= 0 and j >= 0:
        if board[i][j] == 'Q':
            return False
        i -= 1
        j -= 1

    # check right diagonal
    i, j = row, col
    while i >= 0 and j < n:
        if board[i][j] == 'Q':
            return False
        i -= 1
        j += 1

    return True


def backtrack(board, row, n):
    # -------------------------------
    # BACKTRACKING FUNCTION
    # -------------------------------

    if row == n:
        print("\nSolution:")
        for r in board:
            print(' '.join(r))
        return

    for col in range(n):
        # apply bounding condition
        if bound(board, row, col, n):

            # place queen
            board[row][col] = 'Q'

            # recursive call
            backtrack(board, row + 1, n)

            # backtrack (undo)
            board[row][col] = '.'


# -------------------------------
# MAIN
# -------------------------------
n = 4
board = [['.' for _ in range(n)] for _ in range(n)]

print ("back: ")
backtrack(board, 0, n)
